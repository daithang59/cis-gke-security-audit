% ---------------------------------------------------------------------------
% Placeholder figure macro
% Usage: \placeholderfig{filename-without-ext}{width}{height}{description}
% If figures/<filename>.pdf exists  → \includegraphics
% Otherwise                         → framed box with description text
% ---------------------------------------------------------------------------
\newcommand{\placeholderfig}[4]{%
  \IfFileExists{figures/#1.pdf}{%
    \includegraphics[width=#2]{#1}%
  }{%
    \begin{tikzpicture}
      \draw[reportblue, thick, dashed, rounded corners]
        (0,0) rectangle (#2,#3);
      % Description — upper area
      \node[text width=10cm, align=center, font=\small\color{reportblue}]
        at (0.5*#2, 0.65*#3)
        {#4};
      % Status note — middle area
      \node[font=\footnotesize\itshape\color{gray}]
        at (0.5*#2, 0.35*#3)
        {Sơ đồ sẽ được bổ sung};
      % Filename — bottom area
      \node[font=\tiny\ttfamily\color{gray}]
        at (0.5*#2, 0.15*#3)
        {figures/#1.pdf};
    \end{tikzpicture}%
  }%
}
